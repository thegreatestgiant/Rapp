<%*
// Find Rapp root by looking for a folder whose name ends with "Source PDFs" (ignoring numeric prefix)
const allFolders = app.vault.getAllLoadedFiles().filter(f => f.children);
const sourcePdfFolder = allFolders.find(f => f.name.replace(/^\d+\s*/, '') === 'Source PDFs');
const rappRoot = sourcePdfFolder ? sourcePdfFolder.parent.path : '';
function rappPath(subpath) {
  return rappRoot ? `${rappRoot}/${subpath}` : subpath;
}

function getAcademicMetadata() {
  const now = new Date();
  const month = now.getMonth() + 1;
  const day = now.getDate();
  let year = now.getFullYear();
  let season, testTarget;
  
  if (month === 1 && day < 15) {
    season = 'Fall'; year = year - 1; testTarget = 'Final';
  } else if (month < 6 || (month === 6 && day < 15)) {
    season = 'Spring';
    testTarget = (month < 4) ? 'Midterm' : 'Final';
  } else {
    season = 'Fall';
    testTarget = (month < 11 || (month === 11 && day < 20)) ? 'Midterm' : 'Final';
  }
  
  return { season, year: String(year), testTarget };
}

// Load config from the Rapp root
const configFile = app.vault.getAbstractFileByPath(rappPath('config.json'));
let config = { subjects: ['Gemara', 'Halacha', 'Navi', 'Ikarim'], seasons: ['Spring', 'Fall'], test_types: ['Final', 'Midterm'] };
if (configFile) {
  try {
    config = JSON.parse(await app.vault.read(configFile));
  } catch(e) { console.error('Failed to load config.json:', e); }
}

const autoMeta = getAcademicMetadata();

// --- STEP 1: SUBJECT SELECTION ---
const subjects = config.subjects || ["Gemara", "Halacha", "Navi", "Ikarim"];
let selectedSubject = await tp.system.suggester(subjects, subjects, false, "Select Subject");
if (!selectedSubject) return;

// --- STEP 2: SHEET NAME ---
let sheetName = await tp.system.prompt("Name of Source Sheet (e.g., Perek 5 Overview):");
if (!sheetName) return;

// --- STEP 3: TEST METADATA ---
let seasons = config.seasons || ["Spring", "Fall"];
// Prepend auto-detected season if available
if (autoMeta.season) {
    seasons = Array.from(new Set([autoMeta.season, ...seasons]));
}
seasons.push("None");

let selectedSeason = await tp.system.suggester(seasons, seasons, false, "Select Season");
if (!selectedSeason) return;

let selectedYear = autoMeta.year || tp.date.now("YYYY");

let types = config.test_types || ["Final", "Midterm"];
if (autoMeta.testTarget) {
    types = Array.from(new Set([autoMeta.testTarget, ...types]));
}
types.push("None");

let testType = await tp.system.suggester(types, types, false, "Select Test Type");
if (!testType) return;

// --- STEP 4: DYNAMIC FOLDER ROUTING ---
const sheetsPath = rappPath("111 Source Sheets");

// Ensure the folder exists before moving
if (!app.vault.getAbstractFileByPath(sheetsPath)) {
    if (rappRoot && !app.vault.getAbstractFileByPath(rappRoot)) {
        await app.vault.createFolder(rappRoot);
    }
    await app.vault.createFolder(sheetsPath);
}

// --- STEP 5: RENAME AND MOVE ---
let cleanName = sheetName.replace(/[:\\/|?*<>"]/g, '-');
await tp.file.rename(cleanName);
await tp.file.move(`${sheetsPath}/${cleanName}`);

// --- STEP 6: GENERATE CONTENT ---
// Formats frontmatter dynamically to leave blank strings if "None" was selected
let content = `---
subject: "[[${selectedSubject}]]"
season: "${selectedSeason !== "None" ? selectedSeason : ""}"
year: "${selectedYear}"
test_target: "${testType !== "None" ? testType : ""}"
---
# ${sheetName}

## Sources

`;

tR += content;
// Places your typing cursor right on the bullet point under "Linked Sources"
tR += tp.file.cursor(1);
_%>
